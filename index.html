# Script: backup_audit.sh — Automated Server Health, Permissions & Log Rotator
# Author: Uzair Sultan (Linux Administrator)
# ==============================================================================
set -euo pipefail

BACKUP_SRC="/var/www/production_app"
BACKUP_DEST="/var/backups/daily"
TIMESTAMP=$(date +"%Y%m%d_%H%M%S")
ARCHIVE_NAME="app_backup_${TIMESTAMP}.tar.gz"

log_info() {
  echo -e "[INFO $(date +'%T')] $1"
}

perform_backup() {
  log_info "Starting automated archive of ${BACKUP_SRC}..."
  mkdir -p "${BACKUP_DEST}"
  tar -czf "${BACKUP_DEST}/${ARCHIVE_NAME}" -C "${BACKUP_SRC}" .
  log_info "Backup created successfully: ${BACKUP_DEST}/${ARCHIVE_NAME}"
  
  # Rotate archives older than 7 days
  find "${BACKUP_DEST}" -type f -name "*.tar.gz" -mtime +7 -exec rm -f {} \;
  log_info "Old log & backup rotation complete."
}

perform_backup

# binary_tree_dsa.py — Self-Balancing BST & Breadth-First-Search Traversal
# Author: Uzair Sultan (Python Problem Solver)
# ==============================================================================
from collections import deque

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

def level_order_traversal(root: TreeNode) -> list[list[int]]:
    """Time Complexity: O(N), Space Complexity: O(N)"""
    if not root:
        return []
    
    result = []
    queue = deque([root])
    
    while queue:
        level_nodes = []
        for _ in range(len(queue)):
            curr = queue.popleft()
            level_nodes.append(curr.val)
            if curr.left:  queue.append(curr.left)
            if curr.right: queue.append(curr.right)
        result.append(level_nodes)
        
    return result

import ipaddress

def inspect_network(cidr_string: str):
    net = ipaddress.ip_network(cidr_string, strict=False)
    
    print(f"Network Address:    {net.network_address}")
    print(f"Broadcast Address:  {net.broadcast_address}")
    print(f"Netmask:            {net.netmask}")
    print(f"Total Usable Hosts: {net.num_addresses - 2}")
    print(f"Host IP Range:      {net[1]} - {net[-2]}")

# Example Execution for a /24 Enterprise Subnet
